export type StockAdjustment =
  | { mode: "receive"; quantity: number; reason?: string }
  | { mode: "set_balance"; quantity: number; reason: string };

export function nextStockQuantity(current: number, input: StockAdjustment) {
  if (!Number.isInteger(input.quantity) || input.quantity < 0) throw new Error("จำนวนไม่ถูกต้อง");
  if (input.mode === "receive" && input.quantity === 0) throw new Error("จำนวนรับเข้าต้องมากกว่า 0");
  if (input.mode === "set_balance" && !input.reason.trim()) throw new Error("กรุณาระบุเหตุผล");
  return input.mode === "receive" ? current + input.quantity : input.quantity;
}

create or replace function public.adjust_stock_pool(
  p_pool_id uuid, p_mode text, p_quantity integer, p_reason text default null
) returns public.stock_pools language plpgsql security invoker as $$
declare v_pool public.stock_pools; v_before integer; v_after integer;
begin
  select * into v_pool from public.stock_pools where id = p_pool_id for update;
  if v_pool.id is null or not public.auth_user_role_in_store(v_pool.organization_id, v_pool.store_id, 'manager') then
    raise exception 'ไม่มีสิทธิ์จัดการ Stock Pool';
  end if;
  v_before := v_pool.quantity;
  if p_quantity is null or p_quantity < 0 then raise exception 'จำนวนไม่ถูกต้อง'; end if;
  if p_mode = 'receive' then
    if p_quantity = 0 then raise exception 'จำนวนรับเข้าต้องมากกว่า 0'; end if;
    v_after := v_pool.quantity + p_quantity;
  elsif p_mode = 'set_balance' then
    if nullif(btrim(p_reason), '') is null then raise exception 'กรุณาระบุเหตุผล'; end if;
    v_after := p_quantity;
  else raise exception 'โหมดปรับสต๊อกไม่ถูกต้อง'; end if;
  update public.stock_pools set quantity = v_after, updated_at = now() where id = p_pool_id returning * into v_pool;
  insert into public.stock_movements(stock_pool_id,movement_type,quantity_delta,before_quantity,after_quantity,reason,actor_id)
  values(p_pool_id,p_mode,v_after-v_before,v_before,v_after,p_reason,auth.uid());
  return v_pool;
end $$;

export type ApiScope = "products.read" | "inventory.read" | "orders.read";
export async function authenticateApiKey(req: Request, requiredScope?: ApiScope) {
  const plaintext = readKey(req);
  if (!plaintext) return { ok: false as const, status: 401 as const, error: "Missing API key" };
  const supabase = await createSupabaseServiceClient();
  const { data: row } = await supabase.from("api_keys")
    .select("id, organization_id, revoked_at, scopes")
    .eq("key_hash", hashApiKey(plaintext)).maybeSingle();
  if (!row || row.revoked_at) return { ok: false as const, status: 401 as const, error: "Invalid API key" };
  const scopes = row.scopes ?? [];
  if (requiredScope && !scopes.includes(requiredScope)) {
    return { ok: false as const, status: 403 as const, error: "API key ไม่มี scope ที่จำเป็น" };
  }
  return { ok: true as const, organizationId: row.organization_id, apiKeyId: row.id, scopes };
}
